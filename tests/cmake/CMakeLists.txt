cmake_minimum_required(VERSION 3.25)
project(rustxc_cmake_probe LANGUAGES C CXX)
add_library(probe_c STATIC ../native/native/probe.c)
add_library(probe_cpp STATIC ../native/native/probe.cpp)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(NATIVE REQUIRED IMPORTED_TARGET alsa libudev dbus-1 wayland-client x11 xkbcommon)
    target_link_libraries(probe_c PUBLIC PkgConfig::NATIVE)
endif()
add_executable(probe main.cpp)
target_link_libraries(probe PRIVATE probe_c probe_cpp)
